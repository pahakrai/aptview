```typescript
import { Injectable } from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';

@Injectable()
export class MetricCollectorService {
  private metrics: any[] = [];

  // ⭐ IMPORTANT: Collect metrics from EACH execution
  collectExecutionMetrics(execution: {
    reviewId: string;
    config: HITLConfig;
    aiResult: any;
    humanFeedback: any;
    finalDecision: any;
    expectedDecision: any;
    startTime: number;
    endTime: number;
    cost: number;
    success: boolean;
    retryCount: number;
    userRating?: number;
  }) {
    const metrics = {
      id: uuidv4(),
      timestamp: new Date(),
      
      // Quality Metrics
      quality: {
        isCorrect: execution.finalDecision === execution.expectedDecision,
        aiHumanAgreement: execution.aiResult?.decision === execution.humanFeedback?.decision,
        confidence: execution.aiResult?.confidence || 0,
      },
      
      // Operational Metrics
      operational: {
        timeToDecision: execution.endTime - execution.startTime,
        cost: execution.cost,
        retryCount: execution.retryCount,
        success: execution.success,
        source: execution.humanFeedback ? 'human' : 'ai',
      },
      
      // Business Metrics
      business: {
        automation: !execution.humanFeedback,
        humanReview: !!execution.humanFeedback,
        userRating: execution.userRating || 4.5,
        urgent: execution.reviewId.includes('urgent'),
      },
      
      // Config
      config: execution.config,
    };
    
    this.metrics.push(metrics);
    return metrics;
  }
  
  // ⭐ IMPORTANT: Get aggregated metrics for a configuration
  getMetricsForConfig(configId: string) {
    const configMetrics = this.metrics.filter(m => 
      JSON.stringify(m.config) === JSON.stringify(configId)
    );
    
    const n = configMetrics.length;
    if (n === 0) return null;
    
    return {
      // Quality Metrics
      accuracy: configMetrics.filter(m => m.quality.isCorrect).length / n,
      agreement: configMetrics.filter(m => m.quality.aiHumanAgreement).length / 
                 configMetrics.filter(m => m.business.humanReview).length,
      
      // Operational Metrics
      avgTimeToDecision: configMetrics.reduce((s, m) => s + m.operational.timeToDecision, 0) / n,
      avgCost: configMetrics.reduce((s, m) => s + m.operational.cost, 0) / n,
      successRate: configMetrics.filter(m => m.operational.success).length / n,
      
      // Business Metrics
      automationRate: configMetrics.filter(m => m.business.automation).length / n,
      humanReviewRate: configMetrics.filter(m => m.business.humanReview).length / n,
      avgUserRating: configMetrics.reduce((s, m) => s + m.business.userRating, 0) / n,
      
      // Counts
      totalReviews: n,
      humanReviews: configMetrics.filter(m => m.business.humanReview).length,
      aiOnlyReviews: configMetrics.filter(m => m.business.automation).length,
      
      rawMetrics: configMetrics,
    };
  }
  
  // ⭐ IMPORTANT: Reset metrics between experiments
  resetMetrics() {
    this.metrics = [];
  }
}
```

```typescript
import { Injectable } from '@nestjs/common';

@Injectable()
export class CompositeScorerService {
  
  // ⭐ IMPORTANT: Calculate composite score for a configuration
  calculateCompositeScore(
    metrics: any,
    weights: {
      accuracy: number;
      automation: number;
      cost: number;
      satisfaction: number;
      speed: number;
    }
  ): number {
    // Normalize all metrics to 0-1 (higher is better)
    const normalized = {
      accuracy: metrics.accuracy,
      automation: metrics.automationRate,
      cost: Math.max(0, 1 - (metrics.avgCost / 0.10)), // $0.10 max acceptable cost
      satisfaction: metrics.avgUserRating / 5,
      speed: Math.max(0, 1 - (metrics.avgTimeToDecision / 30000)), // 30s max
    };
    
    // Weighted sum
    const score = 
      normalized.accuracy * weights.accuracy +
      normalized.automation * weights.automation +
      normalized.cost * weights.cost +
      normalized.satisfaction * weights.satisfaction +
      normalized.speed * weights.speed;
    
    return Math.max(0, Math.min(1, score));
  }
  
  // ⭐ IMPORTANT: Calculate SVD-derived weights
  calculateSVDWeights(metricsData: number[][]): number[] {
    // For 5 metrics: [accuracy, automation, cost, satisfaction, speed]
    const [U, Σ, Vt] = this.svd(metricsData);
    
    // First principal component weights
    const rawWeights = Vt[0].map(w => Math.abs(w));
    const sum = rawWeights.reduce((a, b) => a + b, 0);
    return rawWeights.map(w => w / sum);
  }
  
  // Simple SVD implementation
  private svd(matrix: number[][]): [number[][], number[], number[][]] {
    // For simplicity, use PCA approximation
    // In production, use mathjs.svd()
    return this.pcaApproximation(matrix);
  }
  
  private pcaApproximation(matrix: number[][]): [number[][], number[], number[][]] {
    // Implementation using covariance matrix
    // This is a placeholder - use actual SVD library in production
    const n = matrix.length;
    const m = matrix[0].length;
    
    // Center the data
    const means = Array(m).fill(0);
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < m; j++) {
        means[j] += matrix[i][j];
      }
    }
    for (let j = 0; j < m; j++) {
      means[j] /= n;
    }
    
    const centered = matrix.map(row => 
      row.map((val, j) => val - means[j])
    );
    
    // Compute covariance matrix
    const cov = Array(m).fill(0).map(() => Array(m).fill(0));
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < m; j++) {
        for (let k = 0; k < m; k++) {
          cov[j][k] += centered[i][j] * centered[i][k];
        }
      }
    }
    for (let j = 0; j < m; j++) {
      for (let k = 0; k < m; k++) {
        cov[j][k] /= n;
      }
    }
    
    // Compute eigenvalues (simplified for demo)
    // In production, use mathjs.eigs()
    const eigenvalues = Array(m).fill(0);
    for (let i = 0; i < m; i++) {
      eigenvalues[i] = cov[i][i];
    }
    
    // Return approximation
    const U = centered;
    const Σ = eigenvalues;
    const Vt = Array(m).fill(0).map(() => Array(m).fill(0));
    for (let i = 0; i < m; i++) {
      Vt[i][i] = 1;
    }
    
    return [U, Σ, Vt];
  }
```

```typescript
// src/statistics/statistical-analyzer.service.ts

import { Injectable } from '@nestjs/common';
import * as ss from 'simple-statistics';

@Injectable()
export class StatisticalAnalyzerService {
  
  // ⭐ IMPORTANT: Calculate t-test, p-value
  tTest(
    controlScores: number[],
    treatmentScores: number[]
  ): {
    pValue: number;
    confidence: number;
    effectSize: number;
    isSignificant: boolean;
  } {
    // Means
    const mean1 = ss.mean(controlScores);
    const mean2 = ss.mean(treatmentScores);
    
    // Standard deviations
    const std1 = ss.standardDeviation(controlScores);
    const std2 = ss.standardDeviation(treatmentScores);
    
    // Sample sizes
    const n1 = controlScores.length;
    const n2 = treatmentScores.length;
    
    // Pooled standard deviation
    const pooledStd = Math.sqrt(
      ((n1 - 1) * std1 ** 2 + (n2 - 1) * std2 ** 2) / (n1 + n2 - 2)
    );
    
    // t-statistic
    const t = (mean1 - mean2) / (pooledStd * Math.sqrt(1/n1 + 1/n2));
    
    // Degrees of freedom
    const df = n1 + n2 - 2;
    
    // p-value (using t-distribution)
    const pValue = this.tDistributionCDF(Math.abs(t), df);
    
    // Cohen's d (effect size)
    const effectSize = (mean1 - mean2) / pooledStd;
    
    return {
      pValue,
      confidence: (1 - pValue) * 100,
      effectSize,
      isSignificant: pValue < 0.05,
    };
  }
  
  // ⭐ IMPORTANT: Calculate confidence intervals
  confidenceInterval(
    scores: number[],
    confidenceLevel: number = 0.95
  ): { lower: number; upper: number; margin: number } {
    const mean = ss.mean(scores);
    const std = ss.standardDeviation(scores);
    const n = scores.length;
    
    // Z-score for confidence level
    const z = this.zScore(confidenceLevel);
    
    // Standard error
    const se = std / Math.sqrt(n);
    
    // Margin of error
    const margin = z * se;
    
    return {
      lower: mean - margin,
      upper: mean + margin,
      margin,
    };
  }
  
  // ⭐ IMPORTANT: Statistical power calculation
  calculatePower(
    controlScores: number[],
    treatmentScores: number[],
    alpha: number = 0.05
  ): number {
    const mean1 = ss.mean(controlScores);
    const mean2 = ss.mean(treatmentScores);
    const std1 = ss.standardDeviation(controlScores);
    const std2 = ss.standardDeviation(treatmentScores);
    const n1 = controlScores.length;
    const n2 = treatmentScores.length;
    
    // Cohen's d
    const pooledStd = Math.sqrt(
      ((n1 - 1) * std1 ** 2 + (n2 - 1) * std2 ** 2) / (n1 + n2 - 2)
    );
    const d = (mean1 - mean2) / pooledStd;
    
    // Power approximation (simplified)
    const zAlpha = 1.96; // For alpha = 0.05 (two-tailed)
    const zBeta = 0.84; // For power = 0.80
    const power = Math.pow(zAlpha - zBeta, 2) * (1 / (1/n1 + 1/n2)) - d ** 2;
    
    return Math.max(0, Math.min(1, power));
  }
  
  // Helper: t-distribution CDF
  private tDistributionCDF(t: number, df: number): number {
    // Approximation using normal distribution for large df
    if (df > 30) {
      return 2 * (1 - this.normalCDF(t));
    }
    // For small df, use approximation
    return Math.exp(-t / (1 + Math.sqrt(df / 2)));
  }
  
  // Helper: normal CDF
  private normalCDF(x: number): number {
    const a1 = 0.254829592;
    const a2 = -0.284496736;
    const a3 = 1.421413741;
    const a4 = -1.453152027;
    const a5 = 1.061405429;
    const p = 0.3275911;
    
    const sign = x < 0 ? -1 : 1;
    x = Math.abs(x);
    const t = 1 / (1 + p * x);
    const y = 1 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x / 2);
    
    return sign < 0 ? 1 - y : y;
  }
  
  // Helper: Z-score
  private zScore(confidence: number): number {
    const zScores = {
      0.90: 1.645,
      0.95: 1.96,
      0.99: 2.576,
    };
    return zScores[confidence] || 1.96;
  }
}
```

```typescript
import { Injectable } from '@nestjs/common';
import { HITLReviewSystem } from '../your-system/hitl-review-system.service';
import { MetricCollectorService } from '../metrics/metric-collector.service';
import { CompositeScorerService } from '../statistics/composite-scorer.service';

@Injectable()
export class ExperimentRunnerService {
  
  constructor(
    private hitlSystem: HITLReviewSystem,
    private metricCollector: MetricCollectorService,
    private compositeScorer: CompositeScorerService,
    private statisticalAnalyzer: StatisticalAnalyzerService,
  ) {}
  
  // ⭐ IMPORTANT: Run a complete experiment
  async runExperiment(
    configId: string,
    config: HITLConfig,
    reviews: any[],
    expectedOutputs: any[]
  ): Promise<any> {
    
    console.log(`📊 Running experiment: ${configId}`);
    
    // Reset metrics for this experiment
    this.metricCollector.resetMetrics();
    
    // Process each review
    for (let i = 0; i < reviews.length; i++) {
      const review = reviews[i];
      const expected = expectedOutputs[i];
      
      const startTime = Date.now();
      
      // ⭐ IMPORTANT: Use your ACTUAL system
      const result = await this.hitlSystem.processReview({
        content: review.content,
        urgency: review.urgency,
        config: config,
      });
      
      const endTime = Date.now();
      
      // ⭐ IMPORTANT: Collect metrics for this execution
      this.metricCollector.collectExecutionMetrics({
        reviewId: review.id,
        config: config,
        aiResult: result.aiResult,
        humanFeedback: result.humanFeedback,
        finalDecision: result.decision,
        expectedDecision: expected,
        startTime,
        endTime,
        cost: result.cost || 0,
        success: result.success,
        retryCount: result.retryCount || 0,
        userRating: result.userRating,
      });
    }
    
    // ⭐ IMPORTANT: Aggregate metrics
    const metrics = this.metricCollector.getMetricsForConfig(config);
    
    // ⭐ IMPORTANT: Calculate composite score
    const compositeScore = this.compositeScorer.calculateCompositeScore(
      metrics,
      this.getWeights()
    );
    
    return {
      configId,
      config,
      metrics,
      compositeScore,
      totalReviews: reviews.length,
    };
  }
  
  // ⭐ IMPORTANT: Run multiple experiments and compare
  async runExperiments(
    controlConfig: HITLConfig,
    experimentConfigs: HITLConfig[],
    reviews: any[],
    expectedOutputs: any[]
  ): Promise<any> {
    
    const results = [];
    
    // Run control
    console.log('🎯 Running control configuration...');
    const controlResult = await this.runExperiment(
      'control',
      controlConfig,
      reviews,
      expectedOutputs
    );
    results.push(controlResult);
    
    // Run experiments
    for (let i = 0; i < experimentConfigs.length; i++) {
      const config = experimentConfigs[i];
      const name = `exp-${String.fromCharCode(65 + i)}`;
      
      console.log(`🎯 Running ${name} configuration...`);
      const result = await this.runExperiment(
        name,
        config,
        reviews,
        expectedOutputs
      );
      results.push(result);
    }
    
    // ⭐ IMPORTANT: Compare results
    return this.compareResults(results);
  }
  
  // ⭐ IMPORTANT: Compare results statistically
  private compareResults(results: any[]) {
    const control = results.find(r => r.configId === 'control');
    const experiments = results.filter(r => r.configId !== 'control');
    
    // Collect composite scores for each configuration
    const scores = results.map(r => ({
      configId: r.configId,
      scores: [r.compositeScore], // In production, you'd have multiple runs
      metrics: r.metrics,
    }));
    
    // Statistical comparisons
    const comparisons = experiments.map(exp => {
      const tTest = this.statisticalAnalyzer.tTest(
        scores.find(s => s.configId === 'control').scores,
        exp.scores
      );
      
      const improvement = ((exp.metrics.accuracy - control.metrics.accuracy) / control.metrics.accuracy) * 100;
      
      return {
        configId: exp.configId,
        compositeScore: exp.compositeScore,
        improvement,
        pValue: tTest.pValue,
        confidence: tTest.confidence,
        effectSize: tTest.effectSize,
        isSignificant: tTest.isSignificant,
        isPractical: Math.abs(tTest.effectSize) > 0.2,
        metrics: exp.metrics,
      };
    });
    
    // Find winner
    const winner = comparisons.find(c => 
      c.isSignificant && 
      c.isPractical && 
      c.improvement > 0
    );
    
    return {
      control: {
        configId: 'control',
        metrics: control.metrics,
        compositeScore: control.compositeScore,
      },
      experiments: comparisons,
      winner: winner ? winner.configId : null,
      improvement: winner ? winner.improvement : 0,
      confidence: winner ? winner.confidence : 0,
      decision: winner ? 'DEPLOY' : 'CONTINUE_TESTING',
    };
  }
  
  private getWeights() {
    return {
      accuracy: 0.30,
      automation: 0.25,
      cost: 0.25,
      satisfaction: 0.15,
      speed: 0.05,
    };
  }
}
```

```typescript
// src/your-system/hitl-review-system.service.ts

import { Injectable } from '@nestjs/common';
import { DeepSeekService } from './deepseek.service';
import { HumanReviewQueue } from './human-review-queue.service';

@Injectable()
export class HITLReviewSystem {
  
  constructor(
    private deepseek: DeepSeekService,
    private humanQueue: HumanReviewQueue,
  ) {}
  
  // ⭐ IMPORTANT: Your actual review processing
  async processReview(params: {
    content: string;
    urgency?: string;
    config: HITLConfig;
  }): Promise<any> {
    
    const { content, urgency, config } = params;
    let retries = 0;
    let aiResult = null;
    let success = false;
    
    // ⭐ IMPORTANT: AI Review with DeepSeek
    while (retries < config.maxRetries && !success) {
      try {
        aiResult = await this.deepseek.review({
          content,
          temperature: config.temperature,
        });
        success = true;
      } catch (error) {
        retries++;
        if (retries >= config.maxRetries) {
          // Queue for human review
          const humanFeedback = await this.humanQueue.queue({
            content,
            urgency,
            aiResult,
            promptType: config.humanPrompt,
          });
          
          return {
            decision: humanFeedback.decision,
            source: 'human',
            success: true,
            retryCount: retries,
            humanFeedback,
            aiResult,
          };
        }
      }
    }
    
    // ⭐ IMPORTANT: Check if human review needed
    if (aiResult.confidence < config.confidenceThreshold) {
      const humanFeedback = await this.humanQueue.queue({
        content,
        urgency,
        aiResult,
        promptType: config.humanPrompt,
      });
      
      return {
        decision: humanFeedback.decision,
        source: 'human',
        success: true,
        retryCount: retries,
        humanFeedback,
        aiResult,
      };
    }
    
    // ⭐ IMPORTANT: Auto-approve
    return {
      decision: aiResult.decision,
      source: 'ai',
      success: true,
      retryCount: retries,
      aiResult,
      confidence: aiResult.confidence,
    };
  }
}
```

### 6.1 Scheduled A/B Tests

```typescript
// src/experiments/experiment-manager.service.ts

import { Injectable } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { ExperimentRunnerService } from './experiment-runner.service';
import { DeploymentManagerService } from '../deployment/deployment-manager.service';

@Injectable()
export class ExperimentManagerService {
  
  constructor(
    private experimentRunner: ExperimentRunnerService,
    private deploymentManager: DeploymentManagerService,
  ) {}
  
  // ⭐ IMPORTANT: Run A/B test on a schedule
  @Cron(CronExpression.EVERY_WEEK) // Run weekly
  async scheduledABTest() {
    console.log('📊 Running scheduled A/B test...');
    
    // Get new test data from the last week
    const testData = await this.getTestData();
    
    // Run experiments
    const result = await this.experimentRunner.runExperiments(
      this.getControlConfig(),
      this.getExperimentConfigs(),
      testData.reviews,
      testData.expectedOutputs
    );
    
    // Log results
    await this.logResults(result);
    
    // Deploy if winner found
    if (result.decision === 'DEPLOY') {
      const winnerConfig = this.getConfigById(result.winner);
      await this.deploymentManager.deployConfiguration(
        winnerConfig,
        result.improvement,
        result.confidence
      );
    }
    
    return result;
  }
  
  // ⭐ IMPORTANT: Get fresh test data
  private async getTestData() {
    // Get reviews from the last 7 days
    const reviews = await this.reviewRepository.find({
      where: { createdAt: MoreThan(new Date(Date.now() - 7 * 24 * 60 * 60 * 1000)) },
      take: 1000,
    });
    
    return {
      reviews: reviews.map(r => ({ id: r.id, content: r.content, urgency: r.urgency })),
      expectedOutputs: reviews.map(r => r.expectedDecision),
    };
  }
}
```
